# Three dkvd processes on loopback, one Raft group per shard (4 shards, replication factor 3).
# The first start of each node initializes its data directory (-init -cluster-id).
$ dkvd -id n1 -listen 127.0.0.1:7001 -peers n2=127.0.0.1:7002,n3=127.0.0.1:7003 -data-dir data/n1 -init -cluster-id demo -cluster -shards 4 -rf 3 -client-listen 127.0.0.1:8001 -admin-listen 127.0.0.1:9001 -metrics-listen 127.0.0.1:9101 >logs/n1.log 2>&1 &
$ dkvd -id n2 -listen 127.0.0.1:7002 -peers n1=127.0.0.1:7001,n3=127.0.0.1:7003 -data-dir data/n2 -init -cluster-id demo -cluster -shards 4 -rf 3 -client-listen 127.0.0.1:8002 -admin-listen 127.0.0.1:9002 -metrics-listen 127.0.0.1:9102 >logs/n2.log 2>&1 &
$ dkvd -id n3 -listen 127.0.0.1:7003 -peers n1=127.0.0.1:7001,n2=127.0.0.1:7002 -data-dir data/n3 -init -cluster-id demo -cluster -shards 4 -rf 3 -client-listen 127.0.0.1:8003 -admin-listen 127.0.0.1:9003 -metrics-listen 127.0.0.1:9103 >logs/n3.log 2>&1 &
